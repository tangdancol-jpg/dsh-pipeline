# 还原方法

```powershell
$src = '%USERPROFILE%\.dsh\profiles\desktop\.pipeline'
$dst = '%USERPROFILE%\.dsh\profiles\desktop\.pipeline-archive'
$all = @(
  # 第一批
  'T-20261014-hello01','T-20261014-hello02','T-20261014-p02v01','T-20261011-5f31b2',
  # 第二批（用户追加）
  'T-20261011-207ede','T-20261011-63633e'

  Move-Item -LiteralPath (Join-Path $dst $n) -Destination $src
}
```
